# Board test executables link the same objects as the core and can call the
# internal scheduler and raster entry points without adding libretro exports.
function(seedless_compute_test target source)
    add_executable(${target} EXCLUDE_FROM_ALL ${source} commands.c context_linux.c dispatch.c
        $<TARGET_OBJECTS:recon_fn>)
    foreach(libretro_source IN LISTS SEEDLESS_LIBRETRO_SOURCES)
        if(IS_ABSOLUTE "${libretro_source}")
            target_sources(${target} PRIVATE "${libretro_source}")
        else()
            target_sources(${target} PRIVATE "${PROJECT_SOURCE_DIR}/${libretro_source}")
        endif()
    endforeach()
    target_include_directories(${target} PRIVATE
        $<TARGET_PROPERTY:seedlessds_libretro,INCLUDE_DIRECTORIES>)
    target_compile_definitions(${target} PRIVATE _GNU_SOURCE SEEDLESS_LIBRETRO=1 SEEDLESS_GLES_COMPUTE=1 SEEDLESS_TEST_EGL=1)
    target_compile_options(${target} PRIVATE -fno-strict-aliasing)
    target_link_libraries(${target} PRIVATE $<TARGET_PROPERTY:seedlessds_libretro,LINK_LIBRARIES>)
    if(SEEDLESS_ENABLE_IPO)
        set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
    endif()
endfunction()
